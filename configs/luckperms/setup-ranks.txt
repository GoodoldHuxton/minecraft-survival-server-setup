# ==========================================================================
#  Fulghen Survival - LuckPerms rank setup
#  Paste these commands into the server console, top to bottom.
#  (Lines starting with # are comments; skip them.)
#
#  Hierarchy:  owner > admin > moderator > vip > default (Member)
#  Each rank inherits everything from the rank below it.
#  VIP perks are cosmetic / quality-of-life only. No pay-to-win.
# ==========================================================================

# ---------- Groups, weights, inheritance ----------
lp creategroup vip
lp creategroup moderator
lp creategroup admin
lp creategroup owner
lp group default setweight 10
lp group vip setweight 20
lp group moderator setweight 60
lp group admin setweight 80
lp group owner setweight 100
lp group vip parent add default
lp group moderator parent add vip
lp group admin parent add moderator
lp group owner parent add admin

# ---------- Prefixes (chat, tab list, name tags) ----------
lp group default meta setprefix 10 "&8[&7Member&8] &7"
lp group vip meta setprefix 20 "&8[&6VIP&8] &e"
lp group moderator meta setprefix 60 "&8[&9Moderator&8] &b"
lp group admin meta setprefix 80 "&8[&cAdmin&8] &c"
lp group owner meta setprefix 100 "&8[&4Owner&8] &4"
